sqrt() -> double

res <- sqrt(9.1)
print(paste(res, typeof(res)))
[1] "3.01662062579967 double"

round() double (double-typed integer)

res <- round(3.1415926, digits = 3)
print(paste(res, typeof(res)))
[1] "3.142 double"
